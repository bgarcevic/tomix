# tx
browse, edit, lint, diff, and deploy tabular semantic models
  opt --help, -?, -h, /?, /h  Show help and usage information
  opt --version  Show version information
  opt --model, -m <path>  Path to the semantic model: a TMDL folder, a .bim file, or a model folder
  opt --output-format <format>  Format for data written to stdout: text, json, csv, tmsl (alias: bim), or tmdl (default: text). Availability varies by command.
  opt --error-format <format>  Format for messages written to stderr: text or json (default: text). Unrecognized values fall back to text.
  opt --server, -s <workspace>  Workspace to connect to: a name, a powerbi:// or asazure:// endpoint, or a local address
  opt --database, -d <name>  Name of the semantic model to use on the workspace
  opt --auth <method>  How to authenticate: auto, interactive, spn, or managed-identity (default: auto)
  opt --recent, --recents [<n>]  Pick a model from the recently used list: no value opens a picker, a number picks that entry (1 is the most recent)
  opt --debug  Show the full stack trace on stderr when an unexpected error occurs
  opt --non-interactive  Never prompt for input. When required input is missing, fail with an error saying what to provide.
  opt --yes, -y  Skip confirmation prompts for destructive operations
  opt --quiet, -q  Suppress non-essential output (spinners, progress, hints). Errors and data still print.

# tx add
Create a new object in the model
  arg <path>  Where to create the object, slash-separated: 'Sales/Revenue', 'Products', 'Admin'. DAX form also accepted: "'Sales'[Revenue]". Combine with -t. For relationships, use 'Sales[Key]->Product[Key]' (many side first).
  arg [model]  Optional path to the model; defaults to the active connection
  opt --type, -t <type>  Type of object to create. Supported values: Table, CalcTable, CalcGroup, Measure, CalcColumn, DataColumn, Hierarchy, Level, Calendar, CalcItem, KPI, Partition, MPartition, EntityPartition, PolicyRangePartition, Expression, Function, Perspective, Culture, ProviderDataSource, StructuredDataSource, Role, TablePermission, Member, Relationship. Data sources always require -t (no path keyword infers them).
  opt --expression, -e <expr>  Expression or value for the new object. Use '-' to read from stdin.
  opt --set <name=value>  Set a property on the new object: name=value, e.g. --set formatString="#,0". Repeatable. Names can use dotted paths, bracket indexers, or a DisplayName.
  opt --file <file>  Read the expression from this file
  opt --if-not-exists  Do nothing and exit 0 when the object already exists
  opt --force, -f  Save despite newly introduced validation errors
  opt --overwrite  Allow --save-to to overwrite an existing target
  opt --dry-run  Preview: show the change without saving, staging, or syncing
  opt --save-to <path>  Write the saved model to this path instead of its source (implies --save)
  opt --serialization <format>  How the model is written: tmdl or bim (tmsl and auto also accepted)
  opt --save  Write this command's change back to the model's source. Cannot be combined with --revert or --stage.
  opt --stage  Stage this command's mutation instead of saving it (commit with 'tx stage commit')
  opt --revert  Revert a staged mutation
  opt --no-sync  Skip workspace sync when workspace mode is active
  opt --mode <mode>  Storage mode for the partition: Import, DirectQuery, Dual, DirectLake, Push, or Default
  opt --source <provider>  Provider name for a ProviderDataSource (e.g. System.Data.SqlClient)
  opt --endpoint <address>  Server or endpoint address for a data source connection
  opt --connection-string <string>  Connection string for a ProviderDataSource
  opt --source-table <name>  Source entity or table name for an EntityPartition
  opt --source-database <name>  Source database name for a ProviderDataSource or StructuredDataSource
  opt --source-schema <name>  Source schema name for an EntityPartition
  opt --partition-expression <expr>  M or DAX expression that defines the partition's source
  opt --columns <names>  Comma-separated column names to create on a new table (Table type only)
  opt --source-type <protocol>  Connection protocol for a StructuredDataSource (e.g. tds)
  opt --range-start <date>  Refresh-policy range start for a PolicyRangePartition (yyyy-MM-dd)
  opt --range-end <date>  Refresh-policy range end for a PolicyRangePartition (yyyy-MM-dd)
  opt --range-granularity <unit>  Refresh-policy range granularity for a PolicyRangePartition: Day, Month, Quarter, or Year (default: Day)

# tx bpa run

  opt --save-to <path>  Write the saved model to this path instead of its source (implies --save)

  opt --rule <id>  Run only this rule. Repeatable.
  opt --ci <system>  Print CI log-group commands to stderr for the given system: vsts or github
  opt --trx <file>  Write results to a .trx test-run file at this path
  opt --allow-external-rules  Allow rule URLs found in model annotations to be fetched
  opt --path <pattern>  Limit analysis to matched objects (literal names, wildcards, or paths)
  opt --no-multiline  Show each rule's guidance on one line

  opt --info  Show only info-severity rules (combinable with --errors/--warnings)
  opt --stage  Stage this command's mutation instead of saving it (commit with 'tx stage commit')

# tx connect
Set or show the active connection
  arg [server]  Workspace name, endpoint, or local model path; combine with --local for Power BI Desktop
  arg [database]  Semantic model name (omit on a TTY to pick from the workspace's models)
  opt --workspace, -w [<target>]  Enable workspace mode: mirror saves between the primary source and a secondary target. Primary remote -> pass a local folder or .bim file. Primary local -> pass <server> <database>. Pass -w with no value to pick the target workspace and model interactively.
  opt --local  Attach to a Power BI Desktop instance running on this machine (Windows only)
  opt --remote  Pick a workspace and semantic model interactively from your Power BI tenant (requires a TTY; sign in first with 'tx auth login')
  opt --list  List without connecting: with <server>, the semantic models on that workspace or endpoint; with --local, running Power BI Desktop instances (report, endpoint, database)
  opt --profile, -p <name>  Connect through this saved profile (list them with 'tx profile list')
  opt --clear  Forget the active connection
  opt --force  Allow workspace mode to initialize over a folder that already has content
  opt --workspace-format <format>  How a local workspace is stored on disk (tmdl or bim); detected from the path when omitted
  opt --workspace-auth <method>  How to authenticate the remote side of workspace mode (default: --auth when set, otherwise auto)

# tx deploy
Deploy a semantic model to a workspace
  arg [model]  Optional path to the model to deploy; defaults to the active connection
  opt --profile, -p <name>  Deploy through a saved profile for this run only; see 'tx profile list' and 'tx profile set'
  opt --create-only  Create the target model only if it does not exist yet; fail when it does
  opt --xmla <file>  Write the deployment as a TMSL script to this file instead of deploying. Pass '-' to write to stdout.
  opt --skip-bpa  Skip the BPA gate check for this deploy
  opt --fix-bpa  Apply BPA rule fixes to the model before deploying, where a rule provides one
  opt --bpa-rules <file>  Additional BPA rule files to enforce for this deploy, alongside the built-in ruleset
  opt --bpa-fail-on <level>  Severity threshold for the BPA gate: error or warning (default: error). Applies before the deploy and again after --fix-bpa fixes. Rules that cannot be evaluated count as error-severity findings.
  opt --force  Force deployment, bypassing validation checks
  opt --ci <system>  Print CI log-group commands to stderr for the given system: vsts or github
  opt --dry-run  Preview what would change on the remote target without deploying
  opt --deploy-connections  Overwrite the target's data sources (default: keep the target's connection strings)
  opt --deploy-partitions  Overwrite the target's table partitions (default: keep the target's partitions and processed data)
  opt --deploy-policy-partitions  With --deploy-partitions: also overwrite incremental-refresh policy partitions, discarding processed data (default: keep them)
  opt --deploy-shared-expressions  Overwrite the target's shared expressions / M parameters (default: keep the target's values)
  opt --deploy-roles  Overwrite the target's security roles (default: keep the target's roles)
  opt --deploy-role-members  With --deploy-roles: also overwrite role members (default: keep the target's members)
  opt --deploy-full  Overwrite everything, including incremental-refresh partitions (cannot be combined with other --deploy-* flags)
